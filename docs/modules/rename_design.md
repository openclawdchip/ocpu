# OCPU Rename 模块设计文档 - 版本3.0

## 文档信息
- **模块名称**: ocpu_rename (寄存器重命名单元)
- **版本**: OCPU-RV64I-r3p0
- **工艺节点**: 3nm
- **总线宽度**: 512-bit
- **更新日期**: 2026-02-13

---

## 概述

`ocpu_rename` 是 OCPU 处理器的寄存器重命名单元，负责将逻辑寄存器映射到物理寄存器，支持乱序执行和精确异常恢复。

### OCPU 3.0 特性

| 特性 | 配置 |
|------|------|
| **重命名宽度** | 16条指令/周期 |
| **物理整数寄存器** | 512个 |
| **物理浮点寄存器** | 512个 |
| **物理向量寄存器** | 256个 |
| **检查点数量** | 16个 |
| **重排序队列 (RCQ)** | 256深度 |

## 核心模块详细分析

### 1. RAT (Register Alias Table)

```systemverilog
module ocpu_rn_rat #(
    parameter NUM_LREG = 32,
    parameter NUM_PREG = 512,           // 512物理寄存器
    parameter PREG_WIDTH = 9,           // 9-bit索引
    parameter READ_PORTS = 32,          // 16指令 x 2源
    parameter WRITE_PORTS = 16          // 16指令 x 1目的
)(
    input  wire                     clk,
    input  wire                     reset_n,
    
    // 读端口: 查询逻辑寄存器对应的物理寄存器
    input  wire [READ_PORTS-1:0]    rden_i,
    input  wire [READ_PORTS-1:0][4:0] lreg_raddr_i,   // 逻辑寄存器号 (0-31)
    output wire [READ_PORTS-1:0][PREG_WIDTH-1:0] preg_rdata_o, // 物理寄存器号
    
    // 写端口: 更新RAT (目的寄存器)
    input  wire [WRITE_PORTS-1:0]   wren_i,
    input  wire [WRITE_PORTS-1:0][4:0] lreg_waddr_i,
    input  wire [WRITE_PORTS-1:0][PREG_WIDTH-1:0] preg_wdata_i,
    
    // 检查点恢复
    input  wire                     restore_valid_i,
    input  wire [NUM_LREG-1:0][PREG_WIDTH-1:0] restore_data_i
);
```

### 2. 空闲列表 (Free List)

```systemverilog
module ocpu_rn_freelist #(
    parameter NUM_PREG = 512,
    parameter PREG_WIDTH = 9,
    parameter ALLOC_PORTS = 16,         // 16个分配端口
    parameter FREE_PORTS = 16           // 16个释放端口
)(
    input  wire                     clk,
    input  wire                     reset_n,
    
    // 分配端口 (新目的寄存器)
    input  wire [ALLOC_PORTS-1:0]   alloc_valid_i,
    output wire [ALLOC_PORTS-1:0][PREG_WIDTH-1:0] alloc_preg_o,
    output wire [ALLOC_PORTS-1:0]   alloc_ready_o,    // 有空闲寄存器
    
    // 释放端口 (指令提交后释放旧寄存器)
    input  wire [FREE_PORTS-1:0]    free_valid_i,
    input  wire [FREE_PORTS-1:0][PREG_WIDTH-1:0] free_preg_i,
    
    // 状态输出
    output wire [8:0]               free_count_o      // 剩余空闲数 (0-512)
);
```

### 3. 重命名控制 (Rename Control)

```systemverilog
module ocpu_rn_control #(
    parameter RENAME_WIDTH = 16,        // 16条重命名
    parameter NUM_LREG = 32,
    parameter NUM_PREG = 512,
    parameter PREG_WIDTH = 9
)(
    input  wire                     clk,
    input  wire                     reset_n,
    
    // 输入: 解码后指令
    input  wire [RENAME_WIDTH-1:0]  valid_i,
    input  wire [RENAME_WIDTH-1:0][4:0] src0_lreg_i,  // 源0逻辑寄存器
    input  wire [RENAME_WIDTH-1:0][4:0] src1_lreg_i,  // 源1逻辑寄存器
    input  wire [RENAME_WIDTH-1:0][4:0] dst_lreg_i,   // 目的逻辑寄存器
    input  wire [RENAME_WIDTH-1:0][1:0] reg_type_i,   // 寄存器类型 (INT/FP/VEC)
    
    // RAT查询输出
    output wire [RENAME_WIDTH-1:0][PREG_WIDTH-1:0] src0_preg_o,
    output wire [RENAME_WIDTH-1:0][PREG_WIDTH-1:0] src1_preg_o,
    
    // 目的寄存器分配
    output wire [RENAME_WIDTH-1:0][PREG_WIDTH-1:0] dst_preg_o,    // 新物理寄存器
    output wire [RENAME_WIDTH-1:0][PREG_WIDTH-1:0] dst_old_preg_o, // 旧物理寄存器
    
    // 控制输出
    output wire [RENAME_WIDTH-1:0]  ready_o,          // 每条指令重命名完成
    output wire                     stall_o           // 资源不足停顿
);
```

### 4. 检查点管理 (Checkpoint)

```systemverilog
module ocpu_rn_checkpoint #(
    parameter NUM_CHECKPOINTS = 16,     // 16个检查点
    parameter CP_WIDTH = 4,             // 4-bit索引
    parameter NUM_LREG = 32,
    parameter PREG_WIDTH = 9
)(
    input  wire                     clk,
    input  wire                     reset_n,
    
    // 创建检查点 (分支预测时)
    input  wire                     cp_create_valid_i,
    input  wire [NUM_LREG-1:0][PREG_WIDTH-1:0] rat_state_i,  // 当前RAT状态
    output wire [CP_WIDTH-1:0]      cp_id_o,
    
    // 恢复检查点 (分支预测错误时)
    input  wire                     cp_restore_valid_i,
    input  wire [CP_WIDTH-1:0]      cp_restore_id_i,
    output wire [NUM_LREG-1:0][PREG_WIDTH-1:0] rat_restore_o,
    
    // 释放检查点 (分支提交后)
    input  wire                     cp_free_valid_i,
    input  wire [CP_WIDTH-1:0]      cp_free_id_i
);
```

### 5. RCQ (Reorder Commit Queue) - 256深度

```systemverilog
module ocpu_rn_rcq #(
    parameter DEPTH = 256,
    parameter INDEX_WIDTH = 8,
    parameter PREG_WIDTH = 9
)(
    input  wire                     clk,
    input  wire                     reset_n,
    
    // 入队 (重命名时)
    input  wire [15:0]              enqueue_valid_i,   // 最多16条
    input  wire [15:0][PREG_WIDTH-1:0] enqueue_old_dst_i, // 旧目的寄存器
    
    // 出队 (提交时)
    input  wire [15:0]              dequeue_valid_i,
    output wire [15:0][PREG_WIDTH-1:0] dequeue_old_dst_o,
    
    // 刷新 (异常时)
    input  wire                     flush_valid_i,
    input  wire [INDEX_WIDTH-1:0]   flush_count_i      // 需要刷新的项数
);
```

## 重命名流程

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                           重命名流程 (每周期16条)                             │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│  Step 1: 接收解码后指令                                                       │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │  输入: 16条解码后指令                                               │   │
│  │  - 指令类型 (INT/FP/VEC)                                            │   │
│  │  - 源0逻辑寄存器 (5-bit x 16)                                       │   │
│  │  - 源1逻辑寄存器 (5-bit x 16)                                       │   │
│  │  - 目的逻辑寄存器 (5-bit x 16)                                      │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│                              │                                               │
│                              ▼                                               │
│  Step 2: RAT查询                                                              │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │  查询RAT获取源寄存器的物理寄存器号                                  │   │
│  │  - 32个读端口并行查询 (16指令 x 2源)                                │   │
│  │  - 输出: src0_preg[15:0], src1_preg[15:0]                           │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│                              │                                               │
│                              ▼                                               │
│  Step 3: 分配新物理寄存器                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │  从空闲列表分配16个新物理寄存器                                     │   │
│  │  - 检查空闲列表是否有足够寄存器                                     │   │
│  │  - 如果有: 分配成功, 继续                                           │   │
│  │  - 如果没有: 停顿, 等待提交释放寄存器                               │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│                              │                                               │
│                              ▼                                               │
│  Step 4: 更新RAT                                                              │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │  更新RAT: 逻辑目的 → 新物理寄存器                                   │   │
│  │  - 16个写端口并行更新                                               │   │
│  │  - 保存旧的物理寄存器号到RCQ                                        │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│                              │                                               │
│                              ▼                                               │
│  Step 5: 输出重命名后指令                                                     │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │  输出:                                                              │   │
│  │  - 物理源寄存器号 (9-bit x 2 x 16)                                  │   │
│  │  - 物理目的寄存器号 (9-bit x 16)                                    │   │
│  │  - 旧物理目的寄存器号 (用于提交后释放)                              │   │
│  │  - ROB索引 (用于顺序提交)                                           │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```
